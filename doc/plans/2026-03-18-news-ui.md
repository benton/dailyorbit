```typescript
import { useEffect, useState } from 'react';
import { Link, useParams } from 'react-router';
import { ApiError, apiFetch } from '../lib/api.ts';

type NewsItem = {
  id: number;
  newsSourceId: number;
  sourceUrl: string;
  title: string;
  author: string | null;
  publishedAt: string | null;
  fetchedAt: string;
  fullText: string | null;
  summary: string | null;
  summarizedBy: string | null;
  summarizedAt: string | null;
  extractStatus: string;
  companyId: number | null;
  tags: string[];
  createdAt: string;
};

export default function NewsDetail() {
  const { id } = useParams<{ id: string }>();
  const [item, setItem] = useState<NewsItem | null>(null);
  const [loading, setLoading] = useState<boolean>(true);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    if (id == null) return;
    let cancelled: boolean = false;
    apiFetch<NewsItem>(`/api/news/${id}`)
      .then((data: NewsItem) => {
        if (!cancelled) setItem(data);
      })
      .catch((err: unknown) => {
        if (!cancelled) {
          setError(
            err instanceof ApiError ? err.message : 'Failed to load news item',
          );
        }
      })
      .finally(() => {
        if (!cancelled) setLoading(false);
      });
    return () => { cancelled = true; };
  }, [id]);

  if (loading) {
    return <p className="text-sm text-neutral-500">Loading...</p>;
  }
  if (error != null || item == null) {
    return (
      <p className="text-sm text-red-600 dark:text-red-400">
        {error ?? 'News item not found'}
      </p>
    );
  }

  const publishedDate: string | null =
    item.publishedAt != null
      ? new Date(item.publishedAt).toLocaleDateString('en-US', {
          year: 'numeric',
          month: 'long',
          day: 'numeric',
        })
      : null;

  return (
    <div className="space-y-6">
      {/* Breadcrumb */}
      <nav className="text-xs text-neutral-400 dark:text-neutral-500">
        <Link className="hover:text-neutral-600 dark:hover:text-neutral-300" to="/">
          Home
        </Link>
        {' / '}
        <Link className="hover:text-neutral-600 dark:hover:text-neutral-300" to="/news">
          News
        </Link>
        {' / '}
        <span className="text-neutral-600 dark:text-neutral-300">
          {item.title.slice(0, 60)}{item.title.length > 60 ? '…' : ''}
        </span>
      </nav>

      {/* Header */}
      <div>
        <h1 className="text-2xl font-bold text-neutral-900 dark:text-neutral-100">
          {item.title}
        </h1>
        <div className="mt-2 flex flex-wrap items-center gap-3 text-sm text-neutral-500 dark:text-neutral-400">
          {publishedDate != null && <span>{publishedDate}</span>}
          {item.author != null && (
            <>
              <span className="text-neutral-300 dark:text-neutral-600">·</span>
              <span>{item.author}</span>
            </>
          )}
          <span className="text-neutral-300 dark:text-neutral-600">·</span>
          <a
            className="text-blue-600 hover:underline dark:text-blue-400"
            href={item.sourceUrl}
            rel="noopener noreferrer"
            target="_blank"
          >
            Original article ↗
          </a>
        </div>
        {item.tags.length > 0 && (
          <div className="mt-2 flex flex-wrap gap-1">
            {item.tags.map((tag: string) => (
              <span
                className="rounded-full bg-neutral-100 px-2 py-0.5 text-xs text-neutral-600 dark:bg-neutral-800 dark:text-neutral-400"
                key={tag}
              >
                {tag}
              </span>
            ))}
          </div>
        )}
      </div>

      {/* Summary */}
      {item.summary != null && (
        <div className="rounded-lg border border-blue-100 bg-blue-50 p-4 dark:border-blue-900 dark:bg-blue-950">
          <p className="mb-1 text-xs font-semibold uppercase tracking-wide text-blue-600 dark:text-blue-400">
            Summary
          </p>
          <p className="text-sm leading-relaxed text-neutral-800 dark:text-neutral-200">
            {item.summary}
          </p>
        </div>
      )}

      {/* Full Text */}
      {item.fullText != null && (
        <div>
          <h2 className="mb-2 text-sm font-semibold text-neutral-500 dark:text-neutral-400">
            Full Text
          </h2>
          <div className="prose prose-sm max-w-none dark:prose-invert whitespace-pre-wrap text-sm leading-relaxed text-neutral-700 dark:text-neutral-300">
            {item.fullText}
          </div>
        </div>
      )}

      {item.fullText == null && item.summary == null && (
        <p className="text-sm text-neutral-500 dark:text-neutral-400">
          No content extracted yet (status: {item.extractStatus}).
        </p>
      )}
    </div>
  );
}
```

```typescript
import type { ColumnDef } from '@tanstack/react-table';
import { useCallback, useMemo } from 'react';
import { useNavigate } from 'react-router';
import { apiFetch } from '../lib/api.ts';
import { DataTable, type DataTableFetchParams, type PaginatedResponse } from '../ui/DataTable.tsx';

type NewsListItem = {
  id: number;
  title: string;
  author: string | null;
  publishedAt: string | null;
  extractStatus: string;
  newsSourceId: number;
  sourceUrl: string;
};

const newsColumns: Array<ColumnDef<NewsListItem, unknown>> = [
  {
    accessorKey: 'publishedAt',
    cell: ({ getValue }) => {
      const v = getValue() as string | null;
      return v != null ? new Date(v).toLocaleDateString() : '—';
    },
    header: 'Published',
    size: 110,
  },
  {
    accessorKey: 'title',
    header: 'Title',
    size: 450,
    cell: ({ getValue }) => (
      <span className="line-clamp-1">{getValue() as string}</span>
    ),
  },
  {
    accessorKey: 'author',
    cell: ({ getValue }) => (getValue() as string | null) ?? '',
    header: 'Author',
    size: 150,
  },
  {
    accessorKey: 'extractStatus',
    header: 'Status',
    size: 100,
  },
];

export default function NewsPage() {
  const navigate = useNavigate();

  const fetchNews = useCallback(
    (params: DataTableFetchParams): Promise<PaginatedResponse<NewsListItem>> => {
      const parts: Array<string> = [
        `page=${params.page}`,
        `pageSize=${params.pageSize}`,
      ];
      if (params.sort != null) parts.push(`sort=${params.sort}`);
      if (params.order != null) parts.push(`order=${params.order}`);
      return apiFetch<PaginatedResponse<NewsListItem>>(
        `/api/news?${parts.join('&')}`,
      );
    },
    [],
  );

  const onRowClick = useCallback(
    (row: NewsListItem) => {
      void navigate(`/news/${row.id}`);
    },
    [navigate],
  );

  const cols = useMemo(() => newsColumns, []);

  return (
    <div className="space-y-4">
      <h1 className="text-2xl font-bold text-neutral-900 dark:text-neutral-100">News</h1>
      <p className="text-sm text-neutral-500 dark:text-neutral-400">
        Satellite industry news aggregated from RSS feeds.
      </p>
      <DataTable<NewsListItem>
        columns={cols}
        defaultOrder="desc"
        defaultSort="published_at"
        fetchData={fetchNews}
        onRowClick={onRowClick}
      />
    </div>
  );
}
```

```typescript
import { useCallback, useEffect, useState } from 'react';
import { apiFetch } from '../lib/api.ts';

type NewsSource = {
  id: number;
  name: string;
  feedUrl: string;
  category: string;
  enabled: boolean;
  fetchIntervalMinutes: number;
  lastFetchedAt: string | null;
  lastFetchStatus: string | null;
  lastFetchError: string | null;
  createdAt: string;
  updatedAt: string;
};

type NewSourceForm = {
  name: string;
  feedUrl: string;
  category: string;
  enabled: boolean;
  fetchIntervalMinutes: number;
};

const EMPTY_FORM: NewSourceForm = {
  name: '',
  feedUrl: '',
  category: 'industry',
  enabled: true,
  fetchIntervalMinutes: 60,
};

export default function NewsSourcesAdmin() {
  const [sources, setSources] = useState<NewsSource[]>([]);
  const [loading, setLoading] = useState<boolean>(true);
  const [error, setError] = useState<string | null>(null);
  const [showAdd, setShowAdd] = useState<boolean>(false);
  const [addForm, setAddForm] = useState<NewSourceForm>(EMPTY_FORM);
  const [saving, setSaving] = useState<boolean>(false);
  const [editId, setEditId] = useState<number | null>(null);
  const [editForm, setEditForm] = useState<NewSourceForm>(EMPTY_FORM);
  const [fetchingId, setFetchingId] = useState<number | null>(null);

  const fetchSources = useCallback(async () => {
    try {
      const data: NewsSource[] = await apiFetch<NewsSource[]>(
        '/api/admin/news/sources',
      );
      setSources(data);
    } catch (err: unknown) {
      setError(err instanceof Error ? err.message : 'Failed to load sources');
    } finally {
      setLoading(false);
    }
  }, []);

  useEffect(() => {
    void fetchSources();
  }, [fetchSources]);

  const handleAdd = async (): Promise<void> => {
    setSaving(true);
    try {
      await apiFetch<NewsSource>('/api/admin/news/sources', {
        method: 'POST',
        body: JSON.stringify(addForm),
      });
      setShowAdd(false);
      setAddForm(EMPTY_FORM);
      await fetchSources();
    } catch (err: unknown) {
      setError(err instanceof Error ? err.message : 'Failed to add source');
    } finally {
      setSaving(false);
    }
  };

  const handleUpdate = async (id: number): Promise<void> => {
    setSaving(true);
    try {
      await apiFetch<NewsSource>(`/api/admin/news/sources/${id}`, {
        method: 'PUT',
        body: JSON.stringify(editForm),
      });
      setEditId(null);
      await fetchSources();
    } catch (err: unknown) {
      setError(err instanceof Error ? err.message : 'Failed to update source');
    } finally {
      setSaving(false);
    }
  };

  const handleDelete = async (id: number): Promise<void> => {
    if (!confirm('Delete this source? All its news items will remain.')) return;
    try {
      await apiFetch(`/api/admin/news/sources/${id}`, { method: 'DELETE' });
      await fetchSources();
    } catch (err: unknown) {
      setError(err instanceof Error ? err.message : 'Failed to delete source');
    }
  };

  const handleToggleEnabled = async (source: NewsSource): Promise<void> => {
    try {
      await apiFetch<NewsSource>(`/api/admin/news/sources/${source.id}`, {
        method: 'PUT',
        body: JSON.stringify({ enabled: !source.enabled }),
      });
      await fetchSources();
    } catch (err: unknown) {
      setError(err instanceof Error ? err.message : 'Failed to toggle source');
    }
  };

  const handleFetchNow = async (id: number): Promise<void> => {
    setFetchingId(id);
    try {
      await apiFetch(`/api/admin/news/sources/${id}/fetch`, { method: 'POST' });
      await fetchSources();
    } catch (err: unknown) {
      setError(err instanceof Error ? err.message : 'Fetch failed');
    } finally {
      setFetchingId(null);
    }
  };

  const startEdit = (source: NewsSource): void => {
    setEditId(source.id);
    setEditForm({
      name: source.name,
      feedUrl: source.feedUrl,
      category: source.category,
      enabled: source.enabled,
      fetchIntervalMinutes: source.fetchIntervalMinutes,
    });
  };

  if (loading) {
    return <p className="text-sm text-neutral-500">Loading...</p>;
  }

  return (
    <div className="space-y-4">
      <div className="flex items-center justify-between">
        <h2 className="text-xl font-bold text-neutral-900 dark:text-neutral-100">
          News Sources
        </h2>
        <button
          className="rounded-md bg-blue-600 px-3 py-1.5 text-sm text-white hover:bg-blue-700"
          onClick={() => setShowAdd(!showAdd)}
          type="button"
        >
          {showAdd ? 'Cancel' : 'Add Source'}
        </button>
      </div>

      {error != null && (
        <p className="text-sm text-red-600 dark:text-red-400">{error}</p>
      )}

      {/* Add form */}
      {showAdd && (
        <div className="rounded-lg border border-neutral-200 p-4 dark:border-neutral-700">
          <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
            <input
              className="rounded border border-neutral-300 px-2 py-1 text-sm dark:border-neutral-600 dark:bg-neutral-800"
              onChange={(e) => setAddForm({ ...addForm, name: e.target.value })}
              placeholder="Source name"
              value={addForm.name}
            />
            <input
              className="rounded border border-neutral-300 px-2 py-1 text-sm dark:border-neutral-600 dark:bg-neutral-800"
              onChange={(e) => setAddForm({ ...addForm, feedUrl: e.target.value })}
              placeholder="Feed URL"
              value={addForm.feedUrl}
            />
            <select
              className="rounded border border-neutral-300 px-2 py-1 text-sm dark:border-neutral-600 dark:bg-neutral-800"
              onChange={(e) => setAddForm({ ...addForm, category: e.target.value })}
              value={addForm.category}
            >
              <option value="industry">Industry</option>
              <option value="regulatory">Regulatory</option>
              <option value="company">Company</option>
              <option value="general">General</option>
            </select>
            <input
              className="rounded border border-neutral-300 px-2 py-1 text-sm dark:border-neutral-600 dark:bg-neutral-800"
              min="5"
              onChange={(e) => setAddForm({ ...addForm, fetchIntervalMinutes: Number(e.target.value) })}
              placeholder="Fetch interval (min)"
              type="number"
              value={addForm.fetchIntervalMinutes}
            />
          </div>
          <button
            className="mt-3 rounded-md bg-green-600 px-3 py-1 text-sm text-white hover:bg-green-700 disabled:opacity-50"
            disabled={saving || addForm.name.trim() === '' || addForm.feedUrl.trim() === ''}
            onClick={() => void handleAdd()}
            type="button"
          >
            {saving ? 'Saving...' : 'Save'}
          </button>
        </div>
      )}

      {/* Sources table */}
      <div className="overflow-x-auto rounded-lg border border-neutral-200 dark:border-neutral-700">
        <table className="min-w-full text-sm">
          <thead className="bg-neutral-50 text-left text-xs text-neutral-500 dark:bg-neutral-800 dark:text-neutral-400">
            <tr>
              <th className="px-3 py-2">Name</th>
              <th className="px-3 py-2">Category</th>
              <th className="px-3 py-2">Enabled</th>
              <th className="px-3 py-2">Last Fetch</th>
              <th className="px-3 py-2">Status</th>
              <th className="px-3 py-2">Actions</th>
            </tr>
          </thead>
          <tbody className="divide-y divide-neutral-100 dark:divide-neutral-800">
            {sources.map((source: NewsSource) => (
              <tr key={source.id}>
                {editId === source.id ? (
                  <>
                    <td className="px-3 py-2">
                      <input
                        className="w-full rounded border border-neutral-300 px-1 py-0.5 text-sm dark:border-neutral-600 dark:bg-neutral-800"
                        onChange={(e) => setEditForm({ ...editForm, name: e.target.value })}
                        value={editForm.name}
                      />
                    </td>
                    <td className="px-3 py-2">
                      <select
                        className="rounded border border-neutral-300 px-1 py-0.5 text-sm dark:border-neutral-600 dark:bg-neutral-800"
                        onChange={(e) => setEditForm({ ...editForm, category: e.target.value })}
                        value={editForm.category}
                      >
                        <option value="industry">Industry</option>
                        <option value="regulatory">Regulatory</option>
                        <option value="company">Company</option>
                        <option value="general">General</option>
                      </select>
                    </td>
                    <td className="px-3 py-2">
                      <input
                        checked={editForm.enabled}
                        onChange={(e) => setEditForm({ ...editForm, enabled: e.target.checked })}
                        type="checkbox"
                      />
                    </td>
                    <td className="px-3 py-2 text-xs text-neutral-400">—</td>
                    <td className="px-3 py-2 text-xs text-neutral-400">—</td>
                    <td className="px-3 py-2">
                      <button
                        className="mr-2 text-xs text-green-600 hover:underline"
                        disabled={saving}
                        onClick={() => void handleUpdate(source.id)}
                        type="button"
                      >
                        Save
                      </button>
                      <button
                        className="text-xs text-neutral-500 hover:underline"
                        onClick={() => setEditId(null)}
                        type="button"
                      >
                        Cancel
                      </button>
                    </td>
                  </>
                ) : (
                  <>
                    <td className="px-3 py-2 font-medium text-neutral-900 dark:text-neutral-100">
                      {source.name}
                      <div className="truncate text-xs text-neutral-400">{source.feedUrl}</div>
                    </td>
                    <td className="px-3 py-2 text-neutral-500">{source.category}</td>
                    <td className="px-3 py-2">
                      <button
                        className={`rounded-full px-2 py-0.5 text-xs font-medium ${
                          source.enabled
                            ? 'bg-green-100 text-green-700 dark:bg-green-900 dark:text-green-300'
                            : 'bg-neutral-100 text-neutral-500 dark:bg-neutral-800 dark:text-neutral-400'
                        }`}
                        onClick={() => void handleToggleEnabled(source)}
                        type="button"
                      >
                        {source.enabled ? 'On' : 'Off'}
                      </button>
                    </td>
                    <td className="px-3 py-2 text-xs text-neutral-500">
                      {source.lastFetchedAt != null
                        ? new Date(source.lastFetchedAt).toLocaleString()
                        : 'Never'}
                    </td>
                    <td className="px-3 py-2">
                      {source.lastFetchStatus != null && (
                        <span
                          className={`text-xs ${
                            source.lastFetchStatus === 'success'
                              ? 'text-green-600'
                              : source.lastFetchStatus === 'failed'
                                ? 'text-red-600'
                                : 'text-amber-600'
                          }`}
                        >
                          {source.lastFetchStatus}
                        </span>
                      )}
                      {source.lastFetchError != null && (
                        <div
                          className="max-w-[200px] truncate text-xs text-red-400"
                          title={source.lastFetchError}
                        >
                          {source.lastFetchError}
                        </div>
                      )}
                    </td>
                    <td className="px-3 py-2">
                      <button
                        className="mr-2 text-xs text-blue-600 hover:underline disabled:text-neutral-400"
                        disabled={fetchingId === source.id}
                        onClick={() => void handleFetchNow(source.id)}
                        type="button"
                      >
                        {fetchingId === source.id ? 'Fetching...' : 'Fetch now'}
                      </button>
                      <button
                        className="mr-2 text-xs text-neutral-600 hover:underline dark:text-neutral-400"
                        onClick={() => startEdit(source)}
                        type="button"
                      >
                        Edit
                      </button>
                      <button
                        className="text-xs text-red-600 hover:underline"
                        onClick={() => void handleDelete(source.id)}
                        type="button"
                      >
                        Delete
                      </button>
                    </td>
                  </>
                )}
              </tr>
            ))}
          </tbody>
        </table>
      </div>
    </div>
  );
}
```
